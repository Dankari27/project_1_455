16fn main() {
17    let args: Vec<String> = env::args().collect();
18
19    // Do we have one lol file?
20    let input = check_args(&args).unwrap_or_else(|message| {
21        eprintln!("{}", message);
22        process::exit(1);
23    });
24
25    let source = fs::read_to_string(&input).unwrap_or_else(|_| {
26        eprintln!("could not read");
27        process::exit(1);
28    });
29
30    // Exit on any error, so we only get past here if everything worked
31    let mut compiler = LolCompiler::new();
32    compiler.compile(&source);
33
34    // Save the html next to the input file with the same name
35    let output = input.with_extension("html");
36    if fs::write(&output, &compiler.html).is_err() {
37        eprintln!("could not write");
38        process::exit(1);
39    }
40    println!("Compiled '{}' into '{}'.", input.display(), output.display());
41
42    // This skips opening the browser 
43    if env::var_os("LOLCOMPILER_NO_BROWSER").is_none() {
44        open_browser(&output);
45    }
46}

48/// Checks that one argument was given and is the proper file type
49fn check_args(args: &[String]) -> Result<PathBuf, String> {
50    if args.len() != 2 {
51        return Err("error: expected exactly one input file".to_string());
52    }
53    let path = PathBuf::from(&args[1]);
54    match path.extension().and_then(|ext| ext.to_str()) {
55        Some(ext) if ext.eq_ignore_ascii_case("lol") => Ok(path),
56        _ => Err("Usage error: only .lol files can be compiled.".to_string()),
57    }
58}

60/// Opens the html file in the default browser
61fn open_browser(path: &Path) {
62    let full_path = std::path::absolute(path).unwrap_or_else(|_| path.to_path_buf());
63    let mut command = if cfg!(target_os = "windows") {
64        let mut cmd = Command::new("cmd");
65        cmd.args(["/C", "start", ""]);
66        cmd
67    } else if cfg!(target_os = "macos") {
68        Command::new("open")
69    } else {
70        Command::new("xdg-open")
71    };
72    if command.arg(&full_path).spawn().is_err() {
73        eprintln!("the HTML was created, but the browser could not be opened");
74    }
75}
